!alias og embed -title "Outgunned Dice Roller" 
-desc "<drac2>
args = &ARGS&
full_input = "".join(args).lower()
mode = "normal"
raw_dice_str = full_input
saved_face = 0
saved_count = 0
mode_char = ""
if "r" in full_input: mode_char = "r"
elif "f" in full_input: mode_char = "f"
elif "a" in full_input: mode_char = "a"
if mode_char:
    mode = "reroll" if mode_char == 'r' else "free" if mode_char == 'f' else "allin"
    parts_mode = full_input.split(mode_char)
    raw_dice_str = parts_mode[0]
    right_side = parts_mode[1].strip()
    right_digits = [c for c in right_side if c.isdigit()]
    if len(right_digits) >= 2:
        saved_face = int(right_digits[0])
        saved_count = int(right_digits[1])
mod = 0
try:
    expr = raw_dice_str.replace('-', '+-')
    parts_math = expr.split('+')
    for p in parts_math:
        p = p.strip()
        if p: mod += int(p)
except:
    mod = 0
if mode == "normal":
    if mod < 2:
        num_dice = 2
        dice_label = "2 (Min)"
    elif mod >= 9:
        num_dice = 9
        dice_label = "9 (Max)"
    else:
        num_dice = mod
        dice_label = str(num_dice)
else:
    if mod < 1:
        num_dice = 1
        dice_label = "1 (Min)"
    elif mod >= 9:
        num_dice = 9
        dice_label = "9 (Max)"
    else:
        num_dice = mod
        dice_label = str(num_dice)
error_msg = ""
if mode == "reroll" and saved_count < 2:
    error_msg = "**[ERROR]** Standard Re-roll failed! You must have at least 1 *Basic Success* (2 matching dice) from the initial roll."
rolls = []
formatted_rolls = "[]"
if not error_msg:
    rolls = [randint(1, 6) for _ in range(num_dice)]
    rolls.sort()
    formatted_rolls = "[" + ", ".join([f"**{x}**" if x == 1 else str(x) for x in rolls]) + "]"
tot = 0
calculation_status = ""
if not error_msg:
    counts = [0, 0, 0, 0, 0, 0]
    for face in rolls:
        counts[face - 1] += 1
    new_matches_to_saved = counts[saved_face - 1] if saved_face > 0 else 0
    new_max_count = max(counts)
    current_best = max(saved_count + new_matches_to_saved, new_max_count)
    if mode == "normal":
        tot = current_best
        calculation_status = "Initial Roll."
    elif mode == "free":
        if current_best > saved_count:
            tot = current_best
            calculation_status = "Free Re-roll Success! Result improved."
        else:
            tot = saved_count
            calculation_status = "Free Re-roll Failed to improve. Initial score secured (No Penalty)."
    elif mode == "reroll":
        if current_best > saved_count:
            tot = current_best
            calculation_status = "Re-roll Success! Result improved."
        else:
            tot = max(1, saved_count - 1)
            calculation_status = "Re-roll Failed to improve! Penalty incurred: lost 1 success level."
    elif mode == "allin":
        if current_best > saved_count:
            tot = current_best
            calculation_status = "ALL IN SUCCESS! Success level drastically increased!"
        else:
            tot = 0
            calculation_status = "ALL IN FAILED TO IMPROVE! ALL PREVIOUS SUCCESSES COMPLETELY LOST!"
if error_msg:
    ladder = "Invalid Action"
    embed_color = "#FF0000"
    output_text = error_msg
else:
    if tot < 2:
        ladder = "Failure"
        embed_color = "#FF0000"
    elif tot == 2:
        ladder = "Basic Success"
        embed_color = "#FFA500"
    elif tot == 3:
        ladder = "Critical Success"
        embed_color = "#FFFF00"
    elif tot == 4:
        ladder = "Extreme Success"
        embed_color = "#00FF00"
    elif tot == 5:
        ladder = "Impossible Success"
        embed_color = "#00FFFF"
    else:
        ladder = "Jackpot!!!"
        embed_color = "#FF00FF"
    mode_title = mode.upper() if mode != "normal" else "INITIAL ROLL"
    output_text = f'**Mode:** {mode_title}\n**Dice:** {dice_label}\n**Rolls:** {formatted_rolls}\n**Result:** {tot} ({ladder})\n*Note: {calculation_status}*'
</drac2>
{{output_text}}" -color "{{embed_color}}"
-footer "Outgunned Roller v1.0 beta."
